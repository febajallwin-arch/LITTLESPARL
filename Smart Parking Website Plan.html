Step 7 — Create the parking database tables

Supabase Dashboard open pannunga → SQL Editor → New query.

Indha SQL-ai paste panni Run click pannunga:

SQL
create table public.parking_lots (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  address text not null,
  price_per_hour numeric(10,2) not null default 40,
  created_at timestamptz not null default now()
);

create table public.parking_slots (
  id uuid primary key default gen_random_uuid(),
  lot_id uuid not null references public.parking_lots(id),
  slot_number text not null,
  status text not null default 'available'
    check (status in ('available', 'occupied', 'reserved')),
  unique (lot_id, slot_number)
);

create table public.bookings (
  id uuid primary key default gen_random_uuid(),
  booking_code text unique not null,
  customer_name text not null,
  slot_id uuid not null references public.parking_slots(id),
  start_time timestamptz not null default now(),
  end_time timestamptz not null,
  amount numeric(10,2) not null,
  status text not null default 'confirmed'
    check (status in ('confirmed', 'completed', 'cancelled')),
  created_at timestamptz not null default now()
);

insert into public.parking_lots (name, address, price_per_hour)
values
  ('Central City Parking', 'City Centre', 40),
  ('Tech Park Plaza', 'IT Corridor', 50),
  ('Metro Mall Parking', 'Shopping District', 30);

Step 8 — Create 20 parking slots for each location

Supabase Dashboard → SQL Editor → New query open pannunga.

Indha SQL code paste panni Run click pannunga:

SQL
-- Create 20 parking slots for every parking location
insert into public.parking_slots (lot_id, slot_number, status)
select
  lots.id,
  'A' || lpad(slot_num::text, 2, '0'),
  case
    when slot_num in (4, 8, 13, 17) then 'occupied'
    else 'available'
  end
from public.parking_lots lots
cross join generate_series(1, 20) as slot_num
on conflict (lot_id, slot_number) do nothing;

Ippo 3 locations × 20 slots = 60 parking slots create aagum. Sila slots demo-ku occupied-ah irukkum.

VS Code-la src/app/page.tsx open pannunga. Existing code-ai full-ah remove panni, indha code paste pannunga:

TypeScript
"use client";

import { useCallback, useEffect, useState } from "react";
import { supabase } from "@/lib/supabase";
import { CarFront, MapPin, RefreshCw, ParkingCircle } from "lucide-react";

type Lot = {
  id: string;
  name: string;
  address: string;
  price_per_hour: number;
};

type Slot = {
  id: string;
  lot_id: string;
  slot_number: string;
  status: "available" | "occupied" | "reserved";
};

export default function Home() {
  const [lots, setLots] = useState<Lot[]>([]);
  const [slots, setSlots] = useState<Slot[]>([]);
  const [lotId, setLotId] = useState("");
  const [selected, setSelected] = useState("");
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");

  const loadLots = useCallback(async () => {
    const { data, error } = await supabase
      .from("parking_lots")
      .select("*")
      .order("name");

    if (error) {
      setError(error.message);
      setLoading(false);
      return;
    }

    setLots(data ?? []);

    if (data?.length && !lotId) {
      setLotId(data[0].id);
    }
  }, [lotId]);

  const loadSlots = useCallback(async (id: string) => {
    if (!id) return;

    const { data, error } = await supabase
      .from("parking_slots")
      .select("*")
      .eq("lot_id", id)
      .order("slot_number");

    if (error) {
      setError(error.message);
      return;
    }

    setSlots(data ?? []);
    setError("");
  }, []);

  useEffect(() => {
    loadLots();
  }, [loadLots]);

  useEffect(() => {
    if (!lotId) return;

    setLoading(true);
    setSelected("");

    loadSlots(lotId).finally(() => setLoading(false));

    const channel = supabase
      .channel(`parking-slots-${lotId}`)
      .on(
        "postgres_changes",
        {
          event: "*",
          schema: "public",
          table: "parking_slots",
          filter: `lot_id=eq.${lotId}`,
        },
        () => loadSlots(lotId)
      )
      .subscribe();

    return () => {
      supabase.removeChannel(channel);
    };
  }, [lotId, loadSlots]);

  const available = slots.filter(s => s.status === "available").length;
  const occupied = slots.filter(s => s.status === "occupied").length;
  const reserved = slots.filter(s => s.status === "reserved").length;

  const currentLot = lots.find(l => l.id === lotId);

  return (
    <main className="min-h-screen bg-slate-950 text-white">
      <header className="border-b border-white/10 bg-slate-900/80">
        <div className="mx-auto flex max-w-7xl items-center justify-between px-5 py-5">
          <div className="flex items-center gap-3">
            <div className="rounded-xl bg-blue-600 p-2.5">
              <ParkingCircle size={25} />
            </div>
            <div>
              <h1 className="text-xl font-bold">SmartPark</h1>
              <p className="text-xs text-slate-400">SMART CITY PARKING</p>
            </div>
          </div>

          <span className="flex items-center gap-2 text-sm text-emerald-400">
            <span className="h-2 w-2 rounded-full bg-emerald-400" />
            Live parking
          </span>
        </div>
      </header>

      <section className="mx-auto max-w-7xl px-5 py-10">
        <p className="mb-2 text-sm font-semibold uppercase tracking-[0.2em] text-blue-400">
          Smart parking management
        </p>

        <div className="flex flex-col justify-between gap-5 md:flex-row md:items-end">
          <div>
            <h2 className="text-3xl font-bold md:text-4xl">
              Find your parking spot.
            </h2>
            <p className="mt-3 text-slate-400">
              Check availability and select an available slot.
            </p>
          </div>

          <button
            onClick={() => loadSlots(lotId)}
            className="flex items-center justify-center gap-2 rounded-xl border border-white/15 px-4 py-3 text-sm hover:bg-white/5"
          >
            <RefreshCw size={16} />
            Refresh
          </button>
        </div>

        <div className="mt-8 grid gap-4 sm:grid-cols-3">
          <Stat label="Available" value={available} color="text-emerald-400" />
          <Stat label="Occupied" value={occupied} color="text-red-400" />
          <Stat label="Reserved" value={reserved} color="text-amber-400" />
        </div>

        <section className="mt-8 rounded-2xl border border-white/10 bg-slate-900 p-5 md:p-7">
          <div className="flex flex-col gap-4 md:flex-row md:items-center md:justify-between">
            <div>
              <h3 className="text-lg font-semibold">Parking locations</h3>
              <p className="mt-1 text-sm text-slate-400">
                Choose a location to view its slots.
              </p>
            </div>

            <select
              value={lotId}
              onChange={e => setLotId(e.target.value)}
              className="w-full rounded-xl border border-white/10 bg-slate-800 px-4 py-3 text-white outline-none md:max-w-sm"
            >
              {lots.map(lot => (
                <option key={lot.id} value={lot.id}>
                  {lot.name}
                </option>
              ))}
            </select>
          </div>

          {currentLot && (
            <div className="mt-5 flex flex-wrap items-center gap-3 text-sm text-slate-400">
              <MapPin size={16} />
              {currentLot.address}
              <span className="text-blue-400">
                ₹{currentLot.price_per_hour}/hour
              </span>
            </div>
          )}

          <div className="mt-7 flex flex-wrap gap-4 text-xs text-slate-300">
            <Legend color="bg-emerald-500" label="Available" />
            <Legend color="bg-blue-500" label="Selected" />
            <Legend color="bg-red-500" label="Occupied" />
            <Legend color="bg-amber-500" label="Reserved" />
          </div>

          {error && (
            <div className="mt-6 rounded-xl border border-red-500/30 bg-red-500/10 p-4 text-sm text-red-300">
              {error}
            </div>
          )}

          {loading ? (
            <p className="py-16 text-center text-slate-400">
              Loading parking slots...
            </p>
          ) : slots.length === 0 ? (
            <p className="py-16 text-center text-slate-400">
              No slots found for this location.
            </p>
          ) : (
            <>
              <div className="mt-7 grid grid-cols-3 gap-3 sm:grid-cols-4 md:grid-cols-5 lg:grid-cols-10">
                {slots.map(slot => {
                  const isSelected = selected === slot.id;

                  return (
                    <button
                      key={slot.id}
                      disabled={slot.status !== "available"}
                      onClick={() =>
                        setSelected(isSelected ? "" : slot.id)
                      }
                      className={`flex min-h-24 flex-col items-center justify-center gap-2 rounded-xl border p-3 transition
                        ${
                          isSelected
                            ? "border-blue-400 bg-blue-600 text-white ring-2 ring-blue-400/30"
                            : slot.status === "available"
                            ? "border-emerald-500/30 bg-emerald-500/10 text-emerald-300 hover:border-emerald-400"
                            : slot.status === "occupied"
                            ? "cursor-not-allowed border-red-500/20 bg-red-500/10 text-red-300"
                            : "cursor-not-allowed border-amber-500/20 bg-amber-500/10 text-amber-300"
                        }`}
                    >
                      <CarFront size={25} />
                      <span className="font-semibold">{slot.slot_number}</span>
                      <span className="text-[10px] capitalize">
                        {isSelected ? "Selected" : slot.status}
                      </span>
                    </button>
                  );
                })}
              </div>

              <div className="mt-7 rounded-xl bg-slate-800/70 p-4">
                {selected ? (
                  <div className="flex flex-col justify-between gap-4 sm:flex-row sm:items-center">
                    <div>
                      <p className="text-sm text-slate-400">Selected slot</p>
                      <p className="mt-1 text-lg font-bold">
                        {slots.find(s => s.id === selected)?.slot_number}
                      </p>
                    </div>
                    <p className="text-sm text-slate-400">
                      Slot selected. Booking functionality comes next.
                    </p>
                  </div>
                ) : (
                  <p className="text-sm text-slate-400">
                    Select a green slot to continue.
                  </p>
                )}
              </div>
            </>
          )}
        </section>

        <p className="mt-6 text-center text-xs text-slate-500">
          SmartPark · Real-time availability powered by Supabase
        </p>
      </section>
    </main>
  );
}

function Stat({
  label,
  value,
  color,
}: {
  label: string;
  value: number;
  color: string;
}) {
  return (
    <div className="rounded-2xl border border-white/10 bg-slate-900 p-5">
      <p className="text-sm text-slate-400">{label}</p>
      <p className={`mt-2 text-3xl font-bold ${color}`}>{value}</p>
    </div>
  );
}

function Legend({ color, label }: { color: string; label: string }) {
  return (
    <span className="flex items-center gap-2">
      <span className={`h-2.5 w-2.5 rounded-sm ${color}`} />
      {label}
    </span>
  );
}
